* === PONG ===
* click the canvas, then use W and S to move
* the bot plays the right side. first to 5 wins.

background(black)

* paddles and ball
variable(p1y : 500)
variable(p2y : 500)
variable(ballX : 800)
variable(ballY : 500)
variable(ballDX : 12)
variable(ballDY : 8)
variable(p1score : 0)
variable(p2score : 0)
variable(playing : yes)
variable(needsReset : no)

* draw the initial scene
draw(rectangle_1:1:white:noborder:80,500)
draw(rectangle_2:1:white:noborder:1520,500)
draw(circle_1:1:white:noborder:{ballX},{ballY})
draw(text_1"0":7:white:noborder:600,120:pixel)
draw(text_2"0":7:white:noborder:1000,120:pixel)
draw(text_3"PONG":4:silver:noborder:800,80:bungee)

* click to restart after game over
when(click)
  if(playing = no)
    variable(needsReset : yes)
  end()
end()

forever()

  if(needsReset = yes)
    clear()
    variable(p1y : 500)
    variable(p2y : 500)
    variable(ballX : 800)
    variable(ballY : 500)
    variable(ballDX : 12)
    variable(ballDY : 8)
    variable(p1score : 0)
    variable(p2score : 0)
    variable(playing : yes)
    variable(needsReset : no)
    draw(rectangle_1:1:white:noborder:80,500)
    draw(rectangle_2:1:white:noborder:1520,500)
    draw(circle_1:1:white:noborder:{ballX},{ballY})
    draw(text_1"0":7:white:noborder:600,120:pixel)
    draw(text_2"0":7:white:noborder:1000,120:pixel)
    draw(text_3"PONG":4:silver:noborder:800,80:bungee)
  end()

  if(playing)

    * --- player 1 controls (W/S) ---
    if(key = w) then(subtract(p1y, 18))
    if(key = W) then(subtract(p1y, 18))
    if(key = s) then(add(p1y, 18))
    if(key = S) then(add(p1y, 18))
    * keep paddle in bounds
    if(p1y < 80) then(variable(p1y : 80))
    if(p1y > 920) then(variable(p1y : 920))

    * --- bot paddle (follows ball, but slower than ball moves) ---
    if(ballY > p2y) then(add(p2y, 8))
    if(ballY < p2y) then(subtract(p2y, 8))
    if(p2y < 80) then(variable(p2y : 80))
    if(p2y > 920) then(variable(p2y : 920))

    * --- move ball ---
    add(ballX, ballDX)
    add(ballY, ballDY)

    * --- top/bottom bounce ---
    if(ballY <= 20)
      multiply(ballDY, -1)
      beep(C, 40)
    end()
    if(ballY >= 980)
      multiply(ballDY, -1)
      beep(C, 40)
    end()

    * --- left paddle hit (player) ---
    if(ballX <= 130)
      collide(hit, ballX, ballY, 80, p1y, 110)
      if(hit)
        multiply(ballDX, -1)
        variable(ballX : 140)
        beep(E, 60)
      end()
    end()

    * --- right paddle hit (bot) ---
    if(ballX >= 1470)
      collide(hit, ballX, ballY, 1520, p2y, 110)
      if(hit)
        multiply(ballDX, -1)
        variable(ballX : 1460)
        beep(E, 60)
      end()
    end()

    * --- score: ball goes off left side ---
    if(ballX < 0)
      add(p2score, 1)
      variable(ballX : 800)
      variable(ballY : 500)
      multiply(ballDX, -1)
      beep(A, 200)
      delete(text_2)
      draw(text_2"{p2score}":7:white:noborder:1000,120:pixel)
    end()

    * --- score: ball goes off right side ---
    if(ballX > 1600)
      add(p1score, 1)
      variable(ballX : 800)
      variable(ballY : 500)
      multiply(ballDX, -1)
      beep(A, 200)
      delete(text_1)
      draw(text_1"{p1score}":7:white:noborder:600,120:pixel)
    end()

    * --- game over at 5 points ---
    if(p1score >= 5)
      variable(playing : no)
    end()
    if(p2score >= 5)
      variable(playing : no)
    end()

    * --- redraw ---
    delete(rectangle_1)
    draw(rectangle_1:1:white:noborder:80,{p1y})
    delete(rectangle_2)
    draw(rectangle_2:1:white:noborder:1520,{p2y})
    delete(circle_1)
    draw(circle_1:1:white:noborder:{ballX},{ballY})

  end()

  if(playing = no)
    if(needsReset = no)
      if(p1score >= 5)
        draw(text_4"YOU WIN!":8:lime:noborder:800,500:bungee)
      end()
      if(p2score >= 5)
        draw(text_4"BOT WINS":8:tomato:noborder:800,500:bungee)
      end()
      draw(text_5"click to play again":3:silver:noborder:800,620)
      beep(G, 400)
      variable(playing : drawn)
    end()
  end()

  wait(40)
end()
