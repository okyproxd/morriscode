* === TIC-TAC-TOE ===
* click any cell to place your X. the bot plays O.
* three in a row wins!

background(cream)

* board state — list of 9 cells, "-" means empty
list(cells, -, -, -, -, -, -, -, -, -)
variable(turn : X)
variable(winner : none)
variable(needsReset : no)

* draw the grid lines
draw(line_1:3:charcoal:default:600,160:600,840)
draw(line_2:3:charcoal:default:1000,160:1000,840)
draw(line_3:3:charcoal:default:200,400:1400,400)
draw(line_4:3:charcoal:default:200,600:1400,600)

* title
draw(text_t"TIC TAC TOE":4:charcoal:noborder:800,90:bungee)
draw(text_s"your turn (X)":3:navy:noborder:800,930)

* === click to place a piece ===
when(click)
  if(needsReset = no)
    if(winner = none)
      if(turn = X)
        * figure out which cell got clicked
        variable(cellIdx : 0)
        if(clickX < 600)
          if(clickY < 400) then(variable(cellIdx : 1))
          if(clickY >= 400) then(if(clickY < 600) then(variable(cellIdx : 4)))
          if(clickY >= 600) then(variable(cellIdx : 7))
        end()
        if(clickX >= 600)
          if(clickX < 1000)
            if(clickY < 400) then(variable(cellIdx : 2))
            if(clickY >= 400) then(if(clickY < 600) then(variable(cellIdx : 5)))
            if(clickY >= 600) then(variable(cellIdx : 8))
          end()
        end()
        if(clickX >= 1000)
          if(clickY < 400) then(variable(cellIdx : 3))
          if(clickY >= 400) then(if(clickY < 600) then(variable(cellIdx : 6)))
          if(clickY >= 600) then(variable(cellIdx : 9))
        end()
        * place X if cell is empty
        if(cellIdx > 0)
          listGet(curCell, cells, cellIdx)
          if(curCell = -)
            listSet(cells, cellIdx, X)
            variable(turn : O)
            beep(E, 80)
          end()
        end()
      end()
    end()
  end()
  if(needsReset = no)
    if(winner != none)
      variable(needsReset : yes)
    end()
  end()
end()

* === main loop: redraw, check win, do bot turn ===
forever()

  if(needsReset = yes)
    clear()
    list(cells, -, -, -, -, -, -, -, -, -)
    variable(turn : X)
    variable(winner : none)
    variable(needsReset : no)
    draw(line_1:3:charcoal:default:600,160:600,840)
    draw(line_2:3:charcoal:default:1000,160:1000,840)
    draw(line_3:3:charcoal:default:200,400:1400,400)
    draw(line_4:3:charcoal:default:200,600:1400,600)
    draw(text_t"TIC TAC TOE":4:charcoal:noborder:800,90:bungee)
    draw(text_s"your turn (X)":3:navy:noborder:800,930)
    beep(G, 100)
  end()

  * --- redraw all the cells (X or O) ---
  loop(i from 1 to 9)
    delete(text_c{i})
  end()

  loop(i from 1 to 9)
    listGet(c, cells, i)
    if(c != -)
      * compute the cell's pixel center
      variable(col : {i})
      variable(row : {i})
      * row = floor((i-1)/3), col = (i-1) % 3
      subtract(col, 1)
      subtract(row, 1)
      * approximate row/col by dividing
      * row 0 = i in 1,2,3 → subtract gives 0,1,2 → divide by 3 = 0
      * row 1 = i in 4,5,6 → subtract gives 3,4,5 → divide by 3 = 1
      * row 2 = i in 7,8,9 → subtract gives 6,7,8 → divide by 3 = 2
      * we'll use modulo for col
      command(math, col, %, col, 3)
      command(math, row, /, row, 3)
      * floor it
      command(math, row, -, row, 0)
      * row will be a decimal; we want integer. just check ranges instead:
      variable(idx : {i})
      variable(cx : 0)
      variable(cy : 0)
      if(idx = 1) then(variable(cx : 400))
      if(idx = 1) then(variable(cy : 280))
      if(idx = 2) then(variable(cx : 800))
      if(idx = 2) then(variable(cy : 280))
      if(idx = 3) then(variable(cx : 1200))
      if(idx = 3) then(variable(cy : 280))
      if(idx = 4) then(variable(cx : 400))
      if(idx = 4) then(variable(cy : 500))
      if(idx = 5) then(variable(cx : 800))
      if(idx = 5) then(variable(cy : 500))
      if(idx = 6) then(variable(cx : 1200))
      if(idx = 6) then(variable(cy : 500))
      if(idx = 7) then(variable(cx : 400))
      if(idx = 7) then(variable(cy : 720))
      if(idx = 8) then(variable(cx : 800))
      if(idx = 8) then(variable(cy : 720))
      if(idx = 9) then(variable(cx : 1200))
      if(idx = 9) then(variable(cy : 720))

      if(c = X)
        draw(text_c{idx}"X":7:tomato:noborder:{cx},{cy}:bungee)
      end()
      if(c = O)
        draw(text_c{idx}"O":7:dodgerblue:noborder:{cx},{cy}:bungee)
      end()
    end()
  end()

  * --- check for a winner ---
  * 8 winning lines: rows, cols, diagonals
  if(winner = none)
    * row 1
    listGet(a, cells, 1)
    listGet(b, cells, 2)
    listGet(c, cells, 3)
    if(a = b) then(if(b = c) then(if(a != -) then(variable(winner : {a}))))
    * row 2
    listGet(a, cells, 4)
    listGet(b, cells, 5)
    listGet(c, cells, 6)
    if(a = b) then(if(b = c) then(if(a != -) then(variable(winner : {a}))))
    * row 3
    listGet(a, cells, 7)
    listGet(b, cells, 8)
    listGet(c, cells, 9)
    if(a = b) then(if(b = c) then(if(a != -) then(variable(winner : {a}))))
    * col 1
    listGet(a, cells, 1)
    listGet(b, cells, 4)
    listGet(c, cells, 7)
    if(a = b) then(if(b = c) then(if(a != -) then(variable(winner : {a}))))
    * col 2
    listGet(a, cells, 2)
    listGet(b, cells, 5)
    listGet(c, cells, 8)
    if(a = b) then(if(b = c) then(if(a != -) then(variable(winner : {a}))))
    * col 3
    listGet(a, cells, 3)
    listGet(b, cells, 6)
    listGet(c, cells, 9)
    if(a = b) then(if(b = c) then(if(a != -) then(variable(winner : {a}))))
    * diag 1: 1,5,9
    listGet(a, cells, 1)
    listGet(b, cells, 5)
    listGet(c, cells, 9)
    if(a = b) then(if(b = c) then(if(a != -) then(variable(winner : {a}))))
    * diag 2: 3,5,7
    listGet(a, cells, 3)
    listGet(b, cells, 5)
    listGet(c, cells, 7)
    if(a = b) then(if(b = c) then(if(a != -) then(variable(winner : {a}))))
  end()

  * --- check for a draw (board full, no winner) ---
  if(winner = none)
    variable(emptyCount : 0)
    loop(i from 1 to 9)
      listGet(c, cells, i)
      if(c = -) then(add(emptyCount, 1))
    end()
    if(emptyCount = 0)
      variable(winner : draw)
    end()
  end()

  * --- bot's turn ---
  if(winner = none)
    if(turn = O)
      * the bot waits a moment so it feels human-ish
      delete(text_s)
      draw(text_s"bot is thinking...":3:dodgerblue:noborder:800,930)
      wait(600)

      * bot strategy:
      *   1. take a winning move if there is one
      *   2. block X from winning
      *   3. take center
      *   4. take a corner
      *   5. take a side
      variable(botMove : 0)

      * --- 1: find a winning O move ---
      loop(i from 1 to 9)
        if(botMove = 0)
          listGet(c, cells, i)
          if(c = -)
            * try placing O here, see if it wins
            listSet(cells, i, O)
            * check all 8 lines
            variable(found : no)
            * row 1
            listGet(a, cells, 1)
            listGet(b, cells, 2)
            listGet(cc, cells, 3)
            if(a = O) then(if(b = O) then(if(cc = O) then(variable(found : yes))))
            listGet(a, cells, 4)
            listGet(b, cells, 5)
            listGet(cc, cells, 6)
            if(a = O) then(if(b = O) then(if(cc = O) then(variable(found : yes))))
            listGet(a, cells, 7)
            listGet(b, cells, 8)
            listGet(cc, cells, 9)
            if(a = O) then(if(b = O) then(if(cc = O) then(variable(found : yes))))
            listGet(a, cells, 1)
            listGet(b, cells, 4)
            listGet(cc, cells, 7)
            if(a = O) then(if(b = O) then(if(cc = O) then(variable(found : yes))))
            listGet(a, cells, 2)
            listGet(b, cells, 5)
            listGet(cc, cells, 8)
            if(a = O) then(if(b = O) then(if(cc = O) then(variable(found : yes))))
            listGet(a, cells, 3)
            listGet(b, cells, 6)
            listGet(cc, cells, 9)
            if(a = O) then(if(b = O) then(if(cc = O) then(variable(found : yes))))
            listGet(a, cells, 1)
            listGet(b, cells, 5)
            listGet(cc, cells, 9)
            if(a = O) then(if(b = O) then(if(cc = O) then(variable(found : yes))))
            listGet(a, cells, 3)
            listGet(b, cells, 5)
            listGet(cc, cells, 7)
            if(a = O) then(if(b = O) then(if(cc = O) then(variable(found : yes))))
            * undo the test
            listSet(cells, i, -)
            if(found = yes) then(variable(botMove : {i}))
          end()
        end()
      end()

      * --- 2: block X if X is about to win ---
      if(botMove = 0)
        loop(i from 1 to 9)
          if(botMove = 0)
            listGet(c, cells, i)
            if(c = -)
              listSet(cells, i, X)
              variable(found : no)
              listGet(a, cells, 1)
              listGet(b, cells, 2)
              listGet(cc, cells, 3)
              if(a = X) then(if(b = X) then(if(cc = X) then(variable(found : yes))))
              listGet(a, cells, 4)
              listGet(b, cells, 5)
              listGet(cc, cells, 6)
              if(a = X) then(if(b = X) then(if(cc = X) then(variable(found : yes))))
              listGet(a, cells, 7)
              listGet(b, cells, 8)
              listGet(cc, cells, 9)
              if(a = X) then(if(b = X) then(if(cc = X) then(variable(found : yes))))
              listGet(a, cells, 1)
              listGet(b, cells, 4)
              listGet(cc, cells, 7)
              if(a = X) then(if(b = X) then(if(cc = X) then(variable(found : yes))))
              listGet(a, cells, 2)
              listGet(b, cells, 5)
              listGet(cc, cells, 8)
              if(a = X) then(if(b = X) then(if(cc = X) then(variable(found : yes))))
              listGet(a, cells, 3)
              listGet(b, cells, 6)
              listGet(cc, cells, 9)
              if(a = X) then(if(b = X) then(if(cc = X) then(variable(found : yes))))
              listGet(a, cells, 1)
              listGet(b, cells, 5)
              listGet(cc, cells, 9)
              if(a = X) then(if(b = X) then(if(cc = X) then(variable(found : yes))))
              listGet(a, cells, 3)
              listGet(b, cells, 5)
              listGet(cc, cells, 7)
              if(a = X) then(if(b = X) then(if(cc = X) then(variable(found : yes))))
              listSet(cells, i, -)
              if(found = yes) then(variable(botMove : {i}))
            end()
          end()
        end()
      end()

      * --- 3: take center ---
      if(botMove = 0)
        listGet(c, cells, 5)
        if(c = -) then(variable(botMove : 5))
      end()

      * --- 4: take a corner ---
      if(botMove = 0)
        listGet(c, cells, 1)
        if(c = -) then(variable(botMove : 1))
      end()
      if(botMove = 0)
        listGet(c, cells, 3)
        if(c = -) then(variable(botMove : 3))
      end()
      if(botMove = 0)
        listGet(c, cells, 7)
        if(c = -) then(variable(botMove : 7))
      end()
      if(botMove = 0)
        listGet(c, cells, 9)
        if(c = -) then(variable(botMove : 9))
      end()

      * --- 5: take any side ---
      if(botMove = 0)
        listGet(c, cells, 2)
        if(c = -) then(variable(botMove : 2))
      end()
      if(botMove = 0)
        listGet(c, cells, 4)
        if(c = -) then(variable(botMove : 4))
      end()
      if(botMove = 0)
        listGet(c, cells, 6)
        if(c = -) then(variable(botMove : 6))
      end()
      if(botMove = 0)
        listGet(c, cells, 8)
        if(c = -) then(variable(botMove : 8))
      end()

      * place the bot's move
      if(botMove > 0)
        listSet(cells, botMove, O)
        beep(G, 80)
      end()
      variable(turn : X)

      delete(text_s)
      draw(text_s"your turn (X)":3:navy:noborder:800,930)
    end()
  end()

  * --- announce the winner ---
  if(winner != none)
    delete(text_s)
    if(winner = X)
      draw(text_s"YOU WIN!":7:lime:noborder:800,930:bungee)
      beep(C, 200)
      beep(E, 200)
      beep(G, 400)
    end()
    if(winner = O)
      draw(text_s"BOT WINS":7:tomato:noborder:800,930:bungee)
      beep(G, 200)
      beep(E, 200)
      beep(C, 400)
    end()
    if(winner = draw)
      draw(text_s"it's a draw":5:silver:noborder:800,930)
      beep(D, 300)
    end()
    draw(text_s2"click to play again":3:silver:noborder:800,990)
    variable(winner : announced)
  end()

  wait(120)
end()
