* === SNAKE GAME ===
* click the canvas, then use ARROW KEYS to play
* eat the gold dots. don't hit walls or yourself!
* on game over: click anywhere to play again

background(midnight)

* the snake — 3 segments to start, moving right
list(xs, 800, 760, 720)
list(ys, 500, 500, 500)
variable(length : 3)
variable(dx : 40)
variable(dy : 0)
variable(score : 0)
variable(alive : yes)
variable(needsReset : no)

* place the first food at a random spot on the grid
random(foodX, 2, 38)
random(foodY, 2, 23)
multiply(foodX, 40)
multiply(foodY, 40)

* score label and the food
draw(text_1"Score: 0":3:white:noborder:140,50)
draw(circle_99:1:gold:noborder:{foodX},{foodY})

* === Click to restart after game over ===
when(click)
  if(alive = no)
    variable(needsReset : yes)
  end()
end()

* === GAME LOOP ===
forever()

  * handle restart request
  if(needsReset = yes)
    clear()
    list(xs, 800, 760, 720)
    list(ys, 500, 500, 500)
    variable(length : 3)
    variable(dx : 40)
    variable(dy : 0)
    variable(score : 0)
    variable(alive : yes)
    variable(needsReset : no)
    random(foodX, 2, 38)
    random(foodY, 2, 23)
    multiply(foodX, 40)
    multiply(foodY, 40)
    draw(text_1"Score: 0":3:white:noborder:140,50)
    draw(circle_99:1:gold:noborder:{foodX},{foodY})
    beep(G, 100)
  end()

  if(alive)

    * read arrow keys (can't reverse direction)
    if(key = ArrowRight)
      if(dx = 0)
        variable(dx : 40)
        variable(dy : 0)
      end()
    end()
    if(key = ArrowLeft)
      if(dx = 0)
        variable(dx : -40)
        variable(dy : 0)
      end()
    end()
    if(key = ArrowUp)
      if(dy = 0)
        variable(dx : 0)
        variable(dy : -40)
      end()
    end()
    if(key = ArrowDown)
      if(dy = 0)
        variable(dx : 0)
        variable(dy : 40)
      end()
    end()

    * compute new head position
    listGet(oldHeadX, xs, 1)
    listGet(oldHeadY, ys, 1)
    variable(newHeadX : {oldHeadX})
    variable(newHeadY : {oldHeadY})
    add(newHeadX, dx)
    add(newHeadY, dy)

    * walls
    if(newHeadX < 0) then(variable(alive : no))
    if(newHeadX >= 1600) then(variable(alive : no))
    if(newHeadY < 0) then(variable(alive : no))
    if(newHeadY >= 1000) then(variable(alive : no))

    * self-collision
    loop(i from 2 to length)
      listGet(bx, xs, i)
      listGet(by, ys, i)
      if(newHeadX = bx)
        if(newHeadY = by)
          variable(alive : no)
        end()
      end()
    end()

    * food collision
    collide(ate, newHeadX, newHeadY, foodX, foodY, 30)
    if(ate)
      add(score, 1)
      add(length, 1)
      beep(E, 80)
      * grow: duplicate the tail
      listGet(tailX, xs, length)
      listGet(tailY, ys, length)
      listAdd(xs, {tailX})
      listAdd(ys, {tailY})
      * new food
      random(foodX, 2, 38)
      random(foodY, 2, 23)
      multiply(foodX, 40)
      multiply(foodY, 40)
      delete(circle_99)
      draw(circle_99:1:gold:noborder:{foodX},{foodY})
      delete(text_1)
      draw(text_1"Score: {score}":3:white:noborder:140,50)
    end()

    * shift body — each segment takes the previous one's spot
    loop(i from length to 2)
      variable(prev : {i})
      subtract(prev, 1)
      listGet(px, xs, prev)
      listGet(py, ys, prev)
      listSet(xs, i, {px})
      listSet(ys, i, {py})
    end()
    listSet(xs, 1, {newHeadX})
    listSet(ys, 1, {newHeadY})

    * redraw the snake
    loop(i from 1 to length)
      delete(square_{i})
    end()
    loop(i from 1 to length)
      listGet(sx, xs, i)
      listGet(sy, ys, i)
      draw(square_{i}:1:lime:noborder:{sx},{sy})
    end()

  end()

  * game over screen — only draw it once
  if(alive = no)
    if(needsReset = no)
      draw(text_2"GAME OVER":8:tomato:noborder:800,400)
      draw(text_3"Final score: {score}":4:white:noborder:800,520)
      draw(text_4"click anywhere to play again":3:silver:noborder:800,620)
      beep(C, 400)
      variable(alive : drawn)
    end()
  end()

  wait(110)
end()
